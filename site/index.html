Verify it yourself

git clone <this repo> && cd forecast-leaderboard
# 1. find the commit that issued a forecast for some period D
git log --oneline --follow -- data/paris-temp/forecasts.csv
# 2. read the forecast as of that commit
git show <sha>:data/paris-temp/forecasts.csv | grep ',D'
# 3. show that D had no observation in that same tree
git show <sha>:data/paris-temp/observations.csv | grep ',D'   # expect: nothing
# 4. cross-check the commit against the Actions run that produced it
gh api repos/<owner>/<repo>/actions/runs --jq '.workflow_runs[]|{head_sha,created_at,conclusion}'
